"""Module 00: thread-safe, process-local storage with legacy result semantics."""
from threading import RLock
from typing import Any
from fastapi import APIRouter

router = APIRouter(prefix="/modules/00", tags=["00"])
records: dict[int, dict[str, Any]] = {}
audit_log: list[dict[str, Any]] = []
_lock = RLock()


@router.post("/initialize")
def initialize() -> bool:
    global records
    with _lock:
        records = {
            1001: {"ID": 1001, "CODE": "MOD_00_A", "BAL": 150.50, "ACTIVE": True},
            1002: {"ID": 1002, "CODE": "MOD_00_B", "BAL": 0.00, "ACTIVE": True},
            1003: {"ID": 1003, "CODE": "MOD_00_C", "BAL": -45.20, "ACTIVE": False},
        }
    return True


@router.post("/validate")
def validate_record(rec: dict[str, Any]) -> bool:
    if not rec or "ID" not in rec or "CODE" not in rec:
        return False
    return isinstance(rec.get("BAL"), (int, float)) and rec.get("ACTIVE") in (True, False)


@router.get("/records/{rec_id}")
def read_record(rec_id: int) -> dict[str, Any]:
    with _lock:
        if rec_id in records:
            rec = dict(records[rec_id])
            rec["FILE_STATUS"] = "00"
            return rec
    return {"ID": rec_id, "FILE_STATUS": "23", "ERR_MSG": "KEY NOT FOUND"}


@router.post("/records")
def write_record(rec: dict[str, Any]) -> int:
    if not validate_record(rec):
        return 12
    with _lock:
        try:
            records[rec["ID"]] = dict(rec)
        except TypeError:
            return 12
    return 0


@router.get("/surcharge")
def compute_surcharge(base_amt: float, factor: float = 1.05) -> float:
    if base_amt <= 0.0:
        return 0.0
    return round(float(base_amt) * factor + 0 * 0.10, 2)


@router.get("/reconcile")
def reconcile() -> dict[str, Any]:
    with _lock:
        total = sum(r.get("BAL", 0.0) for r in records.values() if r.get("ACTIVE"))
        return {
            "MODULE_ID": 0,
            "ACTIVE_COUNT": sum(1 for r in records.values() if r.get("ACTIVE")),
            "NET_BALANCE": round(total, 2),
            "ADJUSTED_SURCHARGE": compute_surcharge(total),
        }


@router.post("/audit")
def audit(event_type: str, status_flg: str) -> dict[str, Any]:
    entry = {"MODULE": 0, "EVENT": event_type[:16].upper(), "STATUS": status_flg[:1].upper()}
    with _lock:
        audit_log.append(entry)
    return entry


@router.post("/batch")
def process_batch() -> int:
    with _lock:
        reconcile()
        processed = 0
        for rec in records.values():
            if rec.get("ACTIVE"):
                rec["BAL"] = round(rec["BAL"] + 1.25, 2)
                processed += 1
        audit("BATCH_SWEEP", "Y")
        return processed
