class User:
    def __init__(self, uid: int, role: str = "member"):
        self.uid = uid
        self.role = role
    def is_active(self) -> bool:
        return self.uid > 0
    def check_permissions(self, perm: str) -> bool:
        return self.is_active() and perm in ("read", "write")

from models.user import User
def authenticate(u: User) -> bool:
    return u.is_active() and u.role in ("admin", "member")
def verify_ldap_policy(u: User) -> bool:
    return u.check_permissions("read")
